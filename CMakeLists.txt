cmake_minimum_required(VERSION 3.16)
project(cpp-miner LANGUAGES CXX)

# ---------------------------------------------------------------------------
# Global build settings
# ---------------------------------------------------------------------------
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

# Target x86_64 Linux specifically; fail fast on anything else so we
# never silently produce a binary for the wrong ISA.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(WARNING "This project targets Linux; other platforms are unsupported.")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64")
    message(WARNING "This project targets x86_64; building for '${CMAKE_SYSTEM_PROCESSOR}' is unsupported.")
endif()

# Reasonable release optimization flags for a hashing-heavy workload.
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -funroll-loops")

# ---------------------------------------------------------------------------
# Embed config.ini into the binary at build time.
# Reads config.ini from the project root and writes it as a C++ string
# literal inside a generated header. This way, the binary carries its own
# configuration and works standalone on any machine.
# ---------------------------------------------------------------------------
set(EMBEDDED_CONFIG_HEADER "${CMAKE_BINARY_DIR}/generated/EmbeddedConfig.hpp")

# Read config.ini, escape special characters for a C++ string literal,
# and write it into the generated header.
file(READ "${CMAKE_SOURCE_DIR}/config.ini" CONFIG_CONTENT)

# Escape backslashes, double quotes, and newlines for C++ raw string literal.
# Using R"INI(...)" raw literal so escaping is minimal — just ensure no
# )INI appears in the content (extremely unlikely for an INI file).
set(EMBEDDED_CONFIG_DECLARATION
"#pragma once
//
// EmbeddedConfig.hpp
// Auto-generated by CMake at build time. Embeds config.ini directly
// into the binary so the miner works out of the box on any machine
// without an external configuration file.
//
#include <string>

namespace cppminer::config {
/// Returns the config.ini content that was baked into the binary at
/// build time. Used as the fallback default when no file or CLI flags
/// are provided.
inline const std::string& embeddedConfig() {
    static const std::string kConfig = R\"INI(
${CONFIG_CONTENT}
)INI\";
    return kConfig;
}
} // namespace cppminer::config
"
)

file(WRITE "${EMBEDDED_CONFIG_HEADER}" "${EMBEDDED_CONFIG_DECLARATION}")
message(STATUS "Embedded config.ini into ${EMBEDDED_CONFIG_HEADER}")

# ---------------------------------------------------------------------------
# Threading (pthreads, required for std::thread on Linux + affinity APIs)
# ---------------------------------------------------------------------------
set(THREADS_PREFER_PTHREAD_FLAG ON)
find_package(Threads REQUIRED)

# ---------------------------------------------------------------------------
# Source files, grouped to mirror the /include and /src folder layout
# ---------------------------------------------------------------------------
set(SOURCES
    src/main.cpp
    src/config/ConfigManager.cpp
    src/utils/CpuInfo.cpp
    src/utils/MiniJson.cpp
    src/network/StratumClient.cpp
    src/miner/MinerEngine.cpp
    src/hash/Sha256dAlgorithm.cpp
    src/hash/CryptonightAlgorithm.cpp
    src/hash/HashAlgorithmFactory.cpp
    src/threading/ThreadPool.cpp
    src/logger/Logger.cpp
)

add_executable(cpp-miner ${SOURCES})

target_include_directories(cpp-miner PRIVATE
    ${CMAKE_SOURCE_DIR}/include
    ${CMAKE_BINARY_DIR}/generated    # for EmbeddedConfig.hpp
)

target_link_libraries(cpp-miner PRIVATE Threads::Threads)

target_compile_options(cpp-miner PRIVATE
    -Wall -Wextra -Wpedantic
)

# Add dependency so the header is regenerated whenever config.ini changes
set_property(
    DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
    "${CMAKE_SOURCE_DIR}/config.ini"
)

# ---------------------------------------------------------------------------
# Installation (optional convenience: `cmake --install build`)
# ---------------------------------------------------------------------------
install(TARGETS cpp-miner RUNTIME DESTINATION bin)
